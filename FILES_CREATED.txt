To use SEO utilities in your React components:

For Static Pages:
  import { updateMetaTags, SEO_CONFIG } from '@/lib/seo';
  
  export default function AboutPage() {
    useEffect(() => {
      updateMetaTags(SEO_CONFIG.about);
    }, []);
    return (<>{content}</>);
  }

For Blog Posts:
  useEffect(() => {
    const seo = createBlogPostSEO(blogPost);
    updateMetaTags(seo);
  }, [blogPost]);

For Events:
  useEffect(() => {
    const seo = createEventSEO(event);
    updateMetaTags(seo);
  }, [event]);
