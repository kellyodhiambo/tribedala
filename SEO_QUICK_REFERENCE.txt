💻 USAGE IN COMPONENTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

For static pages:
  import { updateMetaTags, SEO_CONFIG } from '@/lib/seo';
  
  useEffect(() => {
    updateMetaTags(SEO_CONFIG.about);
  }, []);

For blog posts:
  const { updateMetaTags, createBlogPostSEO } = require('@/lib/seo');
  
  useEffect(() => {
    updateMetaTags(createBlogPostSEO(blogPost));
  }, [blogPost]);

For events:
  useEffect(() => {
    updateMetaTags({
      ...createEventSEO(event),
      canonicalUrl: `https://tribedala.com/events/${event.id}`
    });
  }, [event]);

🐛 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Sitemap not found (404)?
  → Check /public/sitemap.xml exists
  → Check deployment includes /public files
  → Check web server config

Sitemap not updating?
  → Run: npm run build:sitemap
  → Check .env has SUPABASE credentials
  → Check database has published content

Pages not indexed?
  → Wait 2-4 weeks after submission
  → Submit to Google Search Console
  → Request indexing manually
  → Check robots.txt allows the page

Meta tags not showing?
  → Add useEffect with updateMetaTags()
  → Clear browser cache
  → Check Chrome DevTools → Elements → head
